    Interface BirthWindow

    interface BirthWindow {
        cells: BirthWindowCell[];
        end: Date;
        engineVersion: string;
        flags: BirthWindowFlag[];
        houseSystem: HouseSystem;
        latitude: number;
        longitude: number;
        rounding:
            | {
                end: Date;
                minutes: number;
                mode: "nearest"
                | "down";
                recorded: Date;
                start: Date;
            }
            | null;
        schema: "zodiacs.birth-window.v1";
        start: Date;
        switches: BirthWindowSwitch[];
        unresolved: WindowUnresolved[];
        verification: "sampled at one-second resolution";
    }
